cmake_minimum_required(VERSION 3.20)
project(anosecurekit_enterprise_delivery NONE)

foreach(_required IN ITEMS
    DELIVERY_ROOT
    ENTERPRISE_VERSION
    BACKEND_PROFILE
    DELIVERY_SYSTEM_NAME
    DELIVERY_SYSTEM_PROCESSOR)
  if(NOT DEFINED ${_required} OR "${${_required}}" STREQUAL "")
    message(FATAL_ERROR "${_required} is required")
  endif()
endforeach()

if(NOT IS_DIRECTORY "${DELIVERY_ROOT}")
  message(FATAL_ERROR "DELIVERY_ROOT must name an existing staged install directory")
endif()
if(NOT BACKEND_PROFILE STREQUAL "openssl")
  message(FATAL_ERROR
    "Only the verified openssl Enterprise profile may produce an internal delivery package")
endif()

file(REAL_PATH "${DELIVERY_ROOT}" _delivery_root)
string(TOLOWER "${BACKEND_PROFILE}" _profile_slug)

set(CPACK_PACKAGE_NAME "anosecurekit-enterprise")
set(CPACK_PACKAGE_VENDOR "AnoSecureKit")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY
  "AnoSecureKit Enterprise ${ENTERPRISE_VERSION} ${BACKEND_PROFILE} profile")
set(CPACK_PACKAGE_VERSION "${ENTERPRISE_VERSION}")
set(CPACK_PACKAGE_CONTACT "AnoSecureKit commercial licensing contact")
set(CPACK_PACKAGE_FILE_NAME
  "anosecurekit-enterprise-${ENTERPRISE_VERSION}-${_profile_slug}-${DELIVERY_SYSTEM_NAME}-${DELIVERY_SYSTEM_PROCESSOR}")
set(CPACK_GENERATOR "ZIP;TGZ")
set(CPACK_INCLUDE_TOPLEVEL_DIRECTORY ON)
set(CPACK_MONOLITHIC_INSTALL ON)
set(CPACK_VERBATIM_VARIABLES YES)
set(CPACK_INSTALLED_DIRECTORIES "${_delivery_root};/")
# Enterprise source packages are intentionally not produced. The internal
# delivery contains binaries/headers/docs plus a separate MPL-covered Community
# source-offer archive generated by the parent packaging gate.
set(CPACK_SOURCE_GENERATOR "")
include(CPack)
